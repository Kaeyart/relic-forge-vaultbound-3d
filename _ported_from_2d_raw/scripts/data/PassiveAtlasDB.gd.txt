class_name RVPassiveAtlasDB
extends RefCounted

# Patch 086F: cached passive database.
# The DB must be cheap to query. UI panels call this frequently, so never rebuild
# the tree on every nodes()/node_by_id()/connected_ids() call.

const START_NODE_ID: String = "center"
const CENTER_NODE_ID: String = "center"

static var _nodes_cache: Array[Dictionary] = []
static var _node_map_cache: Dictionary = {}
static var _connections_cache: Dictionary = {}
static var _ordered_cache: Array[String] = []

static func _ensure_cache() -> void:
	if not _nodes_cache.is_empty():
		return
	_build_cache()

	_add_node({"id":"start_vaultbound","name":"Vaultbound Initiate","type":"start","kind":"start","cluster":"core","pos":Vector2(1500,900),"connections":["center"],"cost":0,"stats":{"Maximum Life":8.0,"Maximum Mana":6.0},"rules":[],"tags":["start"],"description":"A neutral starting point for older saves."})

static func _add_class_start(id: String, name: String, pos: Vector2, tags: Array, stats: Dictionary, exits: Array) -> void:
	var conns: Array = ["center"]
	for e: Variant in exits:
		conns.append(str(e))
	_add_node({"id":id,"name":name,"type":"start","kind":"start","cluster":"class","pos":pos,"connections":conns,"cost":0,"stats":stats,"rules":[],"tags":tags,"description":"Class start: " + name})

static func _add_cluster(prefix: String, label: String, base: Vector2, step: Vector2, stat_name: String, tag: String, desc: String, keystone_name: String, rule: String, entry_connections: Array) -> void:
	var previous: String = ""
	for i: int in range(1, 8):
		var id: String = prefix + "_" + str(i).pad_zeros(2)
		var conns: Array = []
		if i == 1:
			for entry: Variant in entry_connections:
				conns.append(str(entry))
		else:
			conns.append(previous)
		var node_type: String = "small"
		var node_name: String = label + " Practice " + str(i)
		var stats: Dictionary = {}
		var rules: Array = []
		var cost: int = 1
		if stat_name.begins_with("Maximum") or stat_name == "Armor":
			stats[stat_name] = 5.0 + float(i) * 2.0
		else:
			stats[stat_name] = 0.025 + float(i) * 0.004
		if i == 4:
			node_type = "notable"
			node_name = label + " Notable"
			stats[stat_name] = stats[stat_name] * 2.0
			cost = 1
		if i == 7:
			node_type = "keystone"
			node_name = keystone_name
			stats = {}
			rules = [rule]
			cost = 1
		_add_node({"id":id,"name":node_name,"type":node_type,"kind":node_type,"cluster":prefix,"pos":base + step * float(i - 1),"connections":conns,"cost":cost,"stats":stats,"rules":rules,"tags":[tag],"description":desc if node_type != "keystone" else "Keystone: " + keystone_name.replace("_", " ")})
		previous = id

static func _add_bridge(id: String, name: String, pos: Vector2, conns: Array, stats: Dictionary, tags: Array, desc: String) -> void:
	_add_node({"id":id,"name":name,"type":"notable","kind":"notable","cluster":"bridge","pos":pos,"connections":conns,"cost":1,"stats":stats,"rules":[],"tags":tags,"description":desc})

static func _add_node(data: Dictionary) -> void:
	var id: String = str(data.get("id", ""))
	if id == "":
		return
	_nodes_cache.append(data)
	_node_map_cache[id] = data
	_ordered_cache.append(id)

static func _rebuild_reverse_connections() -> void:
	_connections_cache.clear()
	for node: Dictionary in _nodes_cache:
		var id: String = str(node.get("id", ""))
		if id == "":
			continue
		if not _connections_cache.has(id):
			_connections_cache[id] = []
		for other_value: Variant in Array(node.get("connections", [])):
			var other: String = str(other_value)
			if other == "":
				continue
			_add_connection(id, other)
			_add_connection(other, id)

static func _add_connection(a: String, b: String) -> void:
	if not _connections_cache.has(a):
		_connections_cache[a] = []
	var arr: Array = Array(_connections_cache[a])
	if not arr.has(b):
		arr.append(b)
	_connections_cache[a] = arr

static func nodes() -> Array[Dictionary]:
	_ensure_cache()
	return _nodes_cache.duplicate(true)

static func all_nodes() -> Array[Dictionary]:
	return nodes()

static func ordered_ids() -> Array[String]:
	_ensure_cache()
	return _ordered_cache.duplicate()

static func ordered_ids_for_class(_class_id: String = "") -> Array[String]:
	return ordered_ids()

static func node_by_id(id: String) -> Dictionary:
	_ensure_cache()
	return Dictionary(_node_map_cache.get(id, {})).duplicate(true)

static func node(id: String) -> Dictionary:
	return node_by_id(id)

static func get_node(id: String) -> Dictionary:
	return node_by_id(id)

static func connected_ids(id: String) -> Array[String]:
	_ensure_cache()
	var out: Array[String] = []
	for value: Variant in Array(_connections_cache.get(id, [])):
		var text: String = str(value)
		if text != "" and not out.has(text):
			out.append(text)
	return out

static func node_count() -> int:
	_ensure_cache()
	return _nodes_cache.size()

static func start_node_id(_class_id: String = "") -> String:
	match str(_class_id):
		"sorceress": return "start_sorceress"
		"warden": return "start_warden"
		"voidbinder": return "start_voidbinder"
		"machinist": return "start_machinist"
	return START_NODE_ID

static func get_start_node_id(class_id: String = "") -> String:
	return start_node_id(class_id)

static func can_allocate(node_id: String, allocated: Array, _class_id: String = "") -> bool:
	if node_id == "":
		return false
	if allocated.has(node_id):
		return false
	var data: Dictionary = node_by_id(node_id)
	if data.is_empty():
		return false
	if str(data.get("type", "")) == "start":
		return true
	for other: String in connected_ids(node_id):
		if allocated.has(other):
			return true
	return false
