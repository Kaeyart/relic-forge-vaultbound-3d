class_name RVClassDB
extends RefCounted

# Patch 086A: class identity foundation.
# This database keeps the old ClassDB API while giving each starter class a
# concrete build direction, starter skills, passive start node, and baseline stats.

const DEFAULT_CLASS_ID: String = "sorceress"

static func ids() -> Array[String]:
	return ["sorceress", "warden", "voidbinder", "machinist"]

static func all_ids() -> Array[String]:
	return ids()

static func default_class_id() -> String:
	return DEFAULT_CLASS_ID

static func has_class(class_id: String) -> bool:
	return _classes().has(class_id)

static func data(class_id: String) -> Dictionary:
	var id: String = class_id
	if not has_class(id):
		id = DEFAULT_CLASS_ID
	return Dictionary(_classes().get(id, {})).duplicate(true)

static func name_for(class_id: String) -> String:
	return str(data(class_id).get("name", class_id.capitalize()))

static func description_for(class_id: String) -> String:
	return str(data(class_id).get("description", ""))

static func identity_for(class_id: String) -> String:
	return str(data(class_id).get("identity", ""))

static func start_node(class_id: String) -> String:
	return str(data(class_id).get("start_node", "center"))

static func starter_skills(class_id: String) -> Array[String]:
	var result: Array[String] = []
	for value: Variant in Array(data(class_id).get("starter_skills", [])):
		var skill_name: String = str(value)
		if skill_name != "" and not result.has(skill_name):
			result.append(skill_name)
	return result

static func tags_for(class_id: String) -> Array[String]:
	var result: Array[String] = []
	for value: Variant in Array(data(class_id).get("tags", [])):
		var tag: String = str(value)
		if tag != "" and not result.has(tag):
			result.append(tag)
	return result

static func rules_for(class_id: String) -> Array[String]:
	var result: Array[String] = []
	for value: Variant in Array(data(class_id).get("rules", [])):
		var rule: String = str(value)
		if rule != "" and not result.has(rule):
			result.append(rule)
	return result

static func stats_for(class_id: String) -> Dictionary:
	return Dictionary(data(class_id).get("stats", {})).duplicate(true)

static func next_class_id(current_id: String) -> String:
	var list: Array[String] = ids()
	if list.is_empty():
		return DEFAULT_CLASS_ID
	var index: int = list.find(current_id)
	if index < 0:
		return list[0]
	return list[(index + 1) % list.size()]

static func class_line(class_id: String) -> String:
	var d: Dictionary = data(class_id)
	return str(d.get("name", class_id)) + " — " + str(d.get("identity", ""))
