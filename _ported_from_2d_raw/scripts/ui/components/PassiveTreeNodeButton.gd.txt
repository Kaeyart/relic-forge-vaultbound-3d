class_name RVPassiveTreeNodeButton
extends Button

signal passive_node_pressed(node_id: String)
signal passive_node_secondary_pressed(node_id: String)

var passive_node_id: String = ""
var passive_state: String = "locked"
var selected: bool = false

func _ready() -> void:
	focus_mode = Control.FOCUS_NONE
	mouse_filter = Control.MOUSE_FILTER_STOP
	custom_minimum_size = Vector2(86.0, 34.0)

func setup(data: Dictionary, state_name: String = "locked", selected_now: bool = false) -> void:
	passive_node_id = str(data.get("id", ""))
	passive_state = state_name
	selected = selected_now
	name = "Node_" + passive_node_id
	text = _short_label(str(data.get("name", passive_node_id)), str(data.get("type", "small")))
	tooltip_text = str(data.get("name", passive_node_id)) + "\n" + str(data.get("description", ""))
	_update_visual()

func set_state(state_name: String, selected_now: bool = false) -> void:
	passive_state = state_name
	selected = selected_now
	_update_visual()

func _short_label(label: String, node_type: String) -> String:
	if node_type == "keystone":
		return "◆ " + label
	if node_type == "notable":
		return "◆ " + label
	if node_type == "start":
		return "● " + label
	return "•"

func _update_visual() -> void:
	var font_color: Color = Color(0.55, 0.48, 0.38, 1.0)
	var bg_color: Color = Color(0.10, 0.075, 0.055, 0.95)
	match passive_state:
		"unlocked":
			font_color = Color(1.0, 0.82, 0.36, 1.0)
			bg_color = Color(0.34, 0.20, 0.08, 0.98)
		"available":
			font_color = Color(0.88, 0.78, 0.54, 1.0)
			bg_color = Color(0.18, 0.125, 0.075, 0.98)
		_:
			font_color = Color(0.42, 0.36, 0.30, 1.0)
			bg_color = Color(0.055, 0.045, 0.04, 0.94)
	if selected:
		font_color = Color(1.0, 0.95, 0.70, 1.0)
		bg_color = Color(0.48, 0.28, 0.10, 1.0)
	add_theme_color_override("font_color", font_color)
	add_theme_color_override("font_hover_color", Color(1.0, 0.95, 0.70, 1.0))
	var style := StyleBoxFlat.new()
	style.bg_color = bg_color
	style.border_color = Color(0.72, 0.44, 0.18, 0.95) if selected else Color(0.34, 0.22, 0.12, 0.78)
	style.border_width_left = 1
	style.border_width_right = 1
	style.border_width_top = 1
	style.border_width_bottom = 1
	style.corner_radius_top_left = 8
	style.corner_radius_top_right = 8
	style.corner_radius_bottom_left = 8
	style.corner_radius_bottom_right = 8
	add_theme_stylebox_override("normal", style)
	add_theme_stylebox_override("hover", style)
	add_theme_stylebox_override("pressed", style)

func _gui_input(event: InputEvent) -> void:
	if event is InputEventMouseButton:
		var mb: InputEventMouseButton = event as InputEventMouseButton
		if mb.pressed and not mb.is_echo():
			if mb.button_index == MOUSE_BUTTON_LEFT:
				passive_node_pressed.emit(passive_node_id)
				accept_event()
			elif mb.button_index == MOUSE_BUTTON_RIGHT:
				passive_node_secondary_pressed.emit(passive_node_id)
				accept_event()
