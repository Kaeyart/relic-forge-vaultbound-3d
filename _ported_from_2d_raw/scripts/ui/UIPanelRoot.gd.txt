class_name RVUIPanelRoot
extends CanvasLayer

# Patch 086G: UI runtime stabilizer.
# Owns panel visibility only. It must not recursively disable panel input.
# It updates the active panel only when the active mode/state signature changes.

var _panel_by_mode: Dictionary = {}
var _last_mode: String = "__unset__"
var _last_signature: String = ""

func _ready() -> void:
	_rebuild_panel_map()
	_apply_mode("")

func _rebuild_panel_map() -> void:
	_panel_by_mode.clear()
	_register_panel("inventory", ["InventoryPanel", "Inventory"])
	_register_panel("crafting", ["CraftingPanel", "Crafting"])
	_register_panel("passive_atlas", ["PassiveAtlasPanel", "PassiveAtlas", "PassiveTreePanel"])
	_register_panel("skill_gems", ["SkillGemsPanel", "SkillGems"])
	_register_panel("stash", ["StashPanel", "Stash"])
	_register_panel("activities", ["ActivityPanel", "ActivitiesPanel", "Activities"])
	_register_panel("map_device", ["MapDevicePanel", "MapDevice"])
	_register_panel("loot_filter", ["LootFilterPanel", "LootFilter"])
	_register_panel("character", ["CharacterPanel", "Character"])

func _register_panel(mode: String, names: Array[String]) -> void:
	for name: String in names:
		var node: Node = find_child(name, true, false)
		if node != null:
			_panel_by_mode[mode] = node
			return

func has_panel_mode(mode: String) -> bool:
	if _panel_by_mode.is_empty():
		_rebuild_panel_map()
	return _panel_by_mode.has(mode)

func update_from_state(state: Object) -> void:
	if _panel_by_mode.is_empty():
		_rebuild_panel_map()
	var mode: String = ""
	if state != null:
		mode = str(state.get("panel_mode"))
	if mode != _last_mode:
		_apply_mode(mode)
		_last_mode = mode
		_last_signature = ""
	var active: Node = _panel_by_mode.get(mode, null)
	if active == null or not active.has_method("update_from_state"):
		return
	var signature: String = _active_signature(state, mode)
	if signature == _last_signature:
		return
	_last_signature = signature
	active.call("update_from_state", state)

func force_refresh() -> void:
	_last_signature = ""

func _apply_mode(mode: String) -> void:
	for key_value: Variant in _panel_by_mode.keys():
		var key: String = str(key_value)
		var panel: Node = _panel_by_mode[key]
		if panel is CanvasItem:
			(panel as CanvasItem).visible = key == mode
		if key == mode:
			_restore_clickable_controls(panel)

func _restore_clickable_controls(root: Node) -> void:
	if root == null:
		return
	if root is BaseButton:
		(root as BaseButton).mouse_filter = Control.MOUSE_FILTER_STOP
	elif root is ScrollContainer:
		(root as ScrollContainer).mouse_filter = Control.MOUSE_FILTER_STOP
	elif root is TextureRect or root is ColorRect or root is Label:
		(root as Control).mouse_filter = Control.MOUSE_FILTER_IGNORE
	elif root is RichTextLabel:
		# Detail text can scroll/select later if needed; for now it must not block buttons.
		(root as RichTextLabel).mouse_filter = Control.MOUSE_FILTER_IGNORE
	elif root is Control:
		var control: Control = root as Control
		if control.mouse_filter == Control.MOUSE_FILTER_IGNORE:
			control.mouse_filter = Control.MOUSE_FILTER_PASS
	for child: Node in root.get_children():
		_restore_clickable_controls(child)

func _active_signature(state: Object, mode: String) -> String:
	if state == null:
		return "null|" + mode
	match mode:
		"inventory":
			return mode + "|" + str(state.get("inventory_cursor")) + "|" + str(state.get("equipment_cursor")) + "|" + str(state.get("backpack")) + "|" + str(state.get("equipped")) + "|" + str(state.get("materials"))
		"crafting":
			return mode + "|" + str(state.get("inventory_cursor")) + "|" + str(state.get("backpack")) + "|" + str(state.get("materials"))
		"passive_atlas":
			return mode + "|" + str(state.get("passive_points")) + "|" + str(state.get("mastery_points")) + "|" + str(state.get("passive_refund_points")) + "|" + str(state.get("refund_points")) + "|" + str(state.get("unlocked_passive_nodes")) + "|" + str(state.get("passive_atlas_allocated"))
		"skill_gems":
			return mode + "|" + str(state.get("skill_gem_cursor")) + "|" + str(state.get("support_gem_cursor")) + "|" + str(state.get("spirit_gem_cursor")) + "|" + str(state.get("skill_gem_inventory")) + "|" + str(state.get("support_gem_inventory")) + "|" + str(state.get("spirit_gem_inventory"))
		"stash":
			return mode + "|" + str(state.get("stash_cursor")) + "|" + str(state.get("stash_tab_mode")) + "|" + str(state.get("stash")) + "|" + str(state.get("map_stash"))
		"map_device":
			return mode + "|" + str(state.get("map_cursor")) + "|" + str(state.get("map_stash"))
		"loot_filter":
			return mode + "|" + str(state.get("loot_filter_preset")) + "|" + str(state.get("loot_filter_settings"))
		"character":
			return mode + "|" + str(state.get("level")) + "|" + str(state.get("character_class_id")) + "|" + str(state.get("build_stats"))
		"activities":
			return mode + "|" + str(state.get("current_activity")) + "|" + str(state.get("rooms_cleared"))
		_:
			return mode
