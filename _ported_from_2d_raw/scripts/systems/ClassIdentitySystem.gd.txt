static func ensure_defaults(state: Object) -> void:
	if state == null:
		return
	var class_id: String = str(_state_get(state, "character_class_id", DEFAULT_CLASS_ID))
	if not CLASS_DATA.has(class_id):
		class_id = DEFAULT_CLASS_ID
		_state_set(state, "character_class_id", class_id)
	var data: Dictionary = Dictionary(CLASS_DATA.get(class_id, CLASS_DATA[DEFAULT_CLASS_ID]))
	_state_set(state, "class_identity_version", 1)
	_state_set(state, "class_display_name", str(data.get("name", class_id.capitalize())))
	_state_set(state, "class_start_node_id", str(data.get("start_node", "start_sorceress")))
	_state_set(state, "class_identity_text", str(data.get("identity", "")))
	_state_set(state, "class_stat_bonuses", Dictionary(data.get("stats", {})).duplicate(true))
	_state_set(state, "class_applied_build_stats", Dictionary(data.get("stats", {})).duplicate(true))
	_state_set(state, "class_tags", _string_array(Array(data.get("tags", []))))
	_state_set(state, "class_rules", _string_array(Array(data.get("rules", []))))
	_state_set(state, "class_starter_skills", _string_array(Array(data.get("starter_skills", []))))
	_ensure_class_start_nodes(state, str(data.get("start_node", "start_sorceress")))

static func apply_starter_skills_if_new_character(state: Object) -> void:
	if state == null:
		return
	ensure_defaults(state)
	var class_id: String = str(_state_get(state, "character_class_id", DEFAULT_CLASS_ID))
	if str(_state_get(state, "class_starter_applied", "")) == class_id:
		return
	var skills: Array[String] = _string_array(Array(_state_get(state, "class_starter_skills", [])))
	if skills.is_empty():
		return
	var unlocked: Array[String] = _string_array(Array(_state_get(state, "unlocked_skills", [])))
	for skill_name: String in skills:
		if not unlocked.has(skill_name):
			unlocked.append(skill_name)
	_state_set(state, "unlocked_skills", unlocked)
	var current_active: Array[String] = _string_array(Array(_state_get(state, "active_skills", [])))
	if current_active.is_empty() or str(_state_get(state, "class_starter_applied", "")) == "":
		_state_set(state, "active_skills", skills.duplicate())
		_state_set(state, "selected_skill_index", 0)
	_state_set(state, "class_starter_applied", class_id)

static func apply_to_state(state: Object) -> void:
	if state == null:
		return
	ensure_defaults(state)
	var build_stats: Dictionary = Dictionary(_state_get(state, "build_stats", {})).duplicate(true)
	var class_stats: Dictionary = Dictionary(_state_get(state, "class_stat_bonuses", {}))
	for key_value: Variant in class_stats.keys():
		var key: String = str(key_value)
		build_stats[key] = float(build_stats.get(key, 0.0)) + float(class_stats[key_value])
	_state_set(state, "build_stats", build_stats)
	var flags: Array[String] = _string_array(Array(_state_get(state, "build_flags", [])))
	for rule_value: Variant in Array(_state_get(state, "class_rules", [])):
		var rule: String = str(rule_value)
		if rule != "" and not flags.has(rule):
			flags.append(rule)
	_state_set(state, "build_flags", flags)

static func class_ids() -> Array[String]:
	return ["sorceress", "warden", "voidbinder", "machinist"]

static func class_data(class_id: String) -> Dictionary:
	if not CLASS_DATA.has(class_id):
		class_id = DEFAULT_CLASS_ID
	return Dictionary(CLASS_DATA[class_id]).duplicate(true)

static func display_name(class_id: String) -> String:
	return str(class_data(class_id).get("name", class_id.capitalize()))

static func next_class_id(current_id: String) -> String:
	var ids: Array[String] = class_ids()
	var index: int = ids.find(current_id)
	if index < 0:
		return ids[0]
	return ids[(index + 1) % ids.size()]

static func _ensure_class_start_nodes(state: Object, start_node_id: String) -> void:
	if start_node_id == "":
		return
	var unlocked: Array[String] = _string_array(Array(_state_get(state, "unlocked_passive_nodes", [])))
	if not unlocked.has(start_node_id):
		unlocked.append(start_node_id)
	_state_set(state, "unlocked_passive_nodes", unlocked)
	var legacy: Array[String] = _string_array(Array(_state_get(state, "passive_atlas_allocated", [])))
	if not legacy.has(start_node_id):
		legacy.append(start_node_id)
	_state_set(state, "passive_atlas_allocated", legacy)

static func _state_get(state: Object, key: String, fallback: Variant = null) -> Variant:
	if state == null:
		return fallback
	var value: Variant = state.get(key)
	return fallback if value == null else value

static func _state_set(state: Object, key: String, value: Variant) -> void:
	if state != null:
		state.set(key, value)

static func _string_array(values: Array) -> Array[String]:
	var out: Array[String] = []
	for value: Variant in values:
		var text: String = str(value)
		if text != "" and not out.has(text):
			out.append(text)
	return out
